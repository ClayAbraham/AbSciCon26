/GeneTools/Metagenomics/kofam_scan/exec_annotation -o 1A_proteins_kofam.tsv -p /GeneTools/Metagenomics/kofam_scan/profiles -k /GeneTools/Metagenomics/kofam_scan/ko_list --cpu 16 --format mapper 1A_proteins.faa

#Run this to generate the uniques file  
cut -f2 proteins_kofam.tsv | grep -v "^$" > ko_list_extracted.txt

#Run this to get a abundance of counts 
 awk -F'\t' '{for(i=1;i<=NF;i++) if($i ~ /^K[0-9][0-9][0-9][0-9][0-9]$/) print $i}' 5A_proteins_kofam.tsv | sort | uniq -c | sort -nr > 5A_ko_counts.txt

#Run this command to get a txt file that will have a map and ko data 
awk 'NR==FNR{a[$2]=$1; next} {gsub("ko:","",$1); gsub("path:","",$2); if($1 in a) path[$2]+=a[$1]} END {for (p in path) print path[p] "\t" p}' ko_counts.txt /your/real/path/ko2pathway.tsv | sort -nr | tee 5A_pathway_abundance.txt

#Output with head should look something like this 
44705 map01100 
44705 ko01100 
21384 map01110 
21384 ko01110 
13119 map01120

#Deduplicate the previous txt file with the following, will generate a txt file with only the map data 
awk '{p=$2; sub(/^ko/,"map",p); path[p]+=$1} END {for (k in path) print path[k] "\t" k}' pathway_abundance.txt | sort -nr > pathway_abundance_dedup.txt

#This provides a number of KO assignments mapped to a particular pathway 
join -1 2 -2 1 <(sort -k2,2 5A_pathway_abundance_dedup.txt) <(awk '{gsub("path:","",$1); print $1 "\t" substr($0, index($0,$2))}' /path/to/project/ko_pathways/pathway_names.tsv | sort -k1,1) > 5A_annotated_pathways.txt

awk '{print $2}' 5A_ko_counts.txt | sort | uniq > 5A_ko_present.txt


sort -k1,1 /path/to/project/ko_pathways/ko2pathway.tsv > ko2pathway.sorted.tsv

awk -F'\t' '{sub(/^ko:/,"",$1); sub(/^path:/,"",$2); print $1"\t"$2}' /path/to/project/ko_pathways/ko2pathway.tsv > ko2pathway.clean.tsv

sort -k1,1 5A_ko_present.txt > 5A_ko_present.sorted.txt
sort -k1,1 ko2pathway.clean.tsv > ko2pathway.clean.sorted.tsv

curl https://rest.kegg.jp/list/pathway > pathway_names.tsv

cut -f2 ko2pathway.clean.tsv | sort | uniq -c | awk '{print $2"\t"$1}' > pathway_total_kos.tsv

awk -F'\t' '{printf "%s\t%s\t%s\t%.2f\n",$1,$2,$3,($2/$3)*100}' 5A_pathway_completeness_raw.tsv > 5A_pathway_completeness.tsv


##
awk -F'\t' '{sub(/^path:map/,"ko",$1); print $1"\t"$2}' pathway_names.tsv > pathway_names.ko.tsv

sort -k1,1 pathway_names.ko.tsv > pathway_names.ko.sorted.tsv

join -t $'\t' -1 1 -2 1 5A_pathway_completeness.sorted.tsv pathway_names.ko.sorted.tsv > 5A_pathway_completeness_annotated.tsv

awk -F'\t' '{printf "%s\t%s\t%s\t%.2f\n",$1,$2,$3,($2/$3)*100}' 5A_pathway_completeness_raw.tsv > 5A_pathway_completeness.tsv
